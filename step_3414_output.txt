228: {
229:     float neighbors[24];

232:     int left_start = (bin - 12 < PIANO_MIN_BIN) ? PIANO_MIN_BIN : bin - 12;
233:     int left_end   = bin - 3;
234:     for (int k = left_start; k <= left_end && count < 24; k++) {
235:         neighbors[count++] = mag[k];
236:     }
237: 
238:     int right_start = bin + 3;
239:     int right_end   = (bin + 12 >= num_bins) ? num_bins - 1 : bin + 12;
240:     for (int k = right_start; k <= right_end && count < 24; k++) {
241:         neighbors[count++] = mag[k];
242:     }
243: 
244:     if (count == 0) return 0.2f;
245: 
246:     // Sắp xếp chèn tìm trung vị
247:     for (int i = 1; i < count; i++) {
248:         float val = neighbors[i];
249:         int j = i - 1;
250:         while (j >= 0 && neighbors[j] > val) {
251:             neighbors[j + 1] = neighbors[j];
252:             j--;
253:         }
254:         neighbors[j + 1] = val;
255:     }
256: 
257:     float median = neighbors[count / 2];
258:     if (median < 0.05f) median = 0.05f;
259:     return median;
